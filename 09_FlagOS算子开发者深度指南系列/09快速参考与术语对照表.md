## 第一章 API快速参考

### 1.1 Triton核心API

```python
# ═══════════════════════════════════════════════════════════════
# Triton语言核心API (triton.language as tl)
# ═══════════════════════════════════════════════════════════════

# ───────────────────────────────────────────────────────────────
# 内存操作
# ───────────────────────────────────────────────────────────────
tl.load(ptr, mask=None, other=0.0)     # 加载数据
tl.store(ptr, value, mask=None)        # 存储数据

# ───────────────────────────────────────────────────────────────
# 数学运算
# ───────────────────────────────────────────────────────────────
tl.abs(x)                              # 绝对值
tl.exp(x)                              # 指数函数
tl.log(x)                              # 自然对数
tl.sqrt(x)                             # 平方根
tl.rsqrt(x)                            # 平方根倒数
tl.sin(x), tl.cos(x)                   # 三角函数
tl.floor(x), tl.ceil(x)                # 取整
tl.maximum(x, y), tl.minimum(x, y)     # 最大/最小值

# ───────────────────────────────────────────────────────────────
# 归约操作
# ───────────────────────────────────────────────────────────────
tl.sum(x, axis=None)                   # 求和
tl.max(x, axis=None)                   # 最大值
tl.min(x, axis=None)                   # 最小值
tl.mean(x, axis=None)                  # 平均值
tl.argmax(x, axis=None)                # 最大值索引
tl.argmin(x, axis=None)                # 最小值索引

# ───────────────────────────────────────────────────────────────
# 矩阵运算
# ───────────────────────────────────────────────────────────────
tl.dot(a, b)                           # 矩阵乘法 (自动使用Tensor Core)
tl.trans(x)                            # 转置

# ───────────────────────────────────────────────────────────────
# 形状操作
# ───────────────────────────────────────────────────────────────
tl.arange(start, end)                  # 创建范围数组
tl.reshape(x, shape)                   # 改变形状
tl.expand_dims(x, axis)                # 扩展维度
tl.squeeze(x)                          # 压缩维度

# ───────────────────────────────────────────────────────────────
# 条件操作
# ───────────────────────────────────────────────────────────────
tl.where(condition, x, y)              # 条件选择
tl.if_then_else(cond, a, b)            # 条件分支

# ───────────────────────────────────────────────────────────────
# 类型转换
# ───────────────────────────────────────────────────────────────
x.to(tl.float32)                       # 转换为float32
x.to(tl.float16)                       # 转换为float16
x.to(tl.int32)                         # 转换为int32

# ───────────────────────────────────────────────────────────────
# 程序标识
# ───────────────────────────────────────────────────────────────
tl.program_id(axis)                    # 获取当前program ID
tl.num_programs(axis)                  # 获取program总数
```

### 1.2 Triton装饰器

```python
# ═══════════════════════════════════════════════════════════════
# Triton装饰器
# ═══════════════════════════════════════════════════════════════

import triton
import triton.language as tl

# ───────────────────────────────────────────────────────────────
# @triton.jit - JIT编译装饰器
# ───────────────────────────────────────────────────────────────
@triton.jit
def my_kernel(x_ptr, y_ptr, n, BLOCK_SIZE: tl.constexpr):
    """将函数编译为GPU kernel"""
    pid = tl.program_id(0)
    offsets = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n
    x = tl.load(x_ptr + offsets, mask=mask)
    tl.store(y_ptr + offsets, x * 2, mask=mask)


# ───────────────────────────────────────────────────────────────
# @triton.autotune - 自动调优装饰器
# ───────────────────────────────────────────────────────────────
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_SIZE': 64}, num_warps=4),
        triton.Config({'BLOCK_SIZE': 128}, num_warps=8),
        triton.Config({'BLOCK_SIZE': 256}, num_warps=8),
    ],
    key=['n'],  # 根据哪些参数选择配置
)
@triton.jit
def autotuned_kernel(x_ptr, y_ptr, n, BLOCK_SIZE: tl.constexpr):
    pass


# ───────────────────────────────────────────────────────────────
# @triton.heuristics - 启发式配置
# ───────────────────────────────────────────────────────────────
@triton.heuristics({
    'BLOCK_SIZE': lambda args: 64 if args['n'] < 1024 else 128,
})
@triton.jit
def heuristic_kernel(x_ptr, y_ptr, n, BLOCK_SIZE: tl.constexpr):
    pass


# ───────────────────────────────────────────────────────────────
# 组合使用
# ───────────────────────────────────────────────────────────────
@triton.autotune(
    configs=[...],
    key=['M', 'N', 'K'],
)
@triton.heuristics({
    'ALLOW_TF32': lambda args: args['M'] * args['N'] * args['K'] > 1024 * 1024,
})
@triton.jit
def combined_kernel(...):
    pass
```

### 1.3 FlagGems API

```python
# ═══════════════════════════════════════════════════════════════
# FlagGems核心API
# ═══════════════════════════════════════════════════════════════

import flaggems

# ───────────────────────────────────────────────────────────────
# 启用/禁用
# ───────────────────────────────────────────────────────────────
flaggems.enable()                      # 启用FlagGems算子替换
flaggems.disable()                     # 禁用FlagGems算子替换
flaggems.is_enabled()                  # 检查是否启用

# ───────────────────────────────────────────────────────────────
# 算子查询
# ───────────────────────────────────────────────────────────────
flaggems.list_available_ops()          # 列出所有可用算子
flaggems.list_registered_ops()         # 列出已注册算子
flaggems.is_op_available(op_name)      # 检查算子是否可用

# ───────────────────────────────────────────────────────────────
# 算子注册
# ───────────────────────────────────────────────────────────────
from flaggems.registration import register_operator

@register_operator("my_op")
def my_op_impl(x: torch.Tensor) -> torch.Tensor:
    """注册自定义算子"""
    return my_triton_kernel(x)
```

### 1.4 FlagTree API

```python
# ═══════════════════════════════════════════════════════════════
# FlagTree核心API
# ═══════════════════════════════════════════════════════════════

import flagtree

# ───────────────────────────────────────────────────────────────
# 后端管理
# ───────────────────────────────────────────────────────────────
flagtree.set_backend(name)             # 设置后端 ('nvidia', 'huawei', 'moore'等)
flagtree.get_backend()                 # 获取当前后端
flagtree.list_backends()               # 列出所有支持的后端
flagtree.with_backend(name)            # 后端上下文管理器

# ───────────────────────────────────────────────────────────────
# 编译
# ───────────────────────────────────────────────────────────────
flagtree.compile(kernel, backend)      # 编译kernel到指定后端
flagtree.compile_for_backends(kernel, backends)  # 多后端编译

# ───────────────────────────────────────────────────────────────
# 自动调优
# ───────────────────────────────────────────────────────────────
from flagtree import autotune

@autotune(
    configs=[...],
    key=['n'],
    backend_specific={
        'huawei': [triton.Config({'BLOCK_SIZE': 64}, ...)],
        'moore': [triton.Config({'BLOCK_SIZE': 32}, ...)],
    }
)
@triton.jit
def multi_backend_kernel(...):
    pass
```

### 1.5 TLE扩展API

```python
# ═══════════════════════════════════════════════════════════════
# TLE (Triton Language Extensions) API
# ═══════════════════════════════════════════════════════════════

from triton.language.extra import tle

# ───────────────────────────────────────────────────────────────
# 高级API
# ───────────────────────────────────────────────────────────────
tle.reduce(tensor, axis, op)           # 归约操作
tle.scan(tensor, axis, op)             # 扫描操作
tle.matmul_block(a, b)                 # 矩阵乘法块

# ───────────────────────────────────────────────────────────────
# 中级API
# ───────────────────────────────────────────────────────────────
tle.warp_reduce(tensor, op)            # Warp级归约
tle.block_reduce(tensor, op)           # Block级归约
tle.load_tile(ptr, shape, offsets, ...)  # 加载矩阵块
tle.store_tile(ptr, tile, offsets, ...)  # 存储矩阵块

# ───────────────────────────────────────────────────────────────
# 归约操作类型
# ───────────────────────────────────────────────────────────────
tle.RedOp.SUM                          # 求和
tle.RedOp.MAX                          # 最大值
tle.RedOp.MIN                          # 最小值
tle.RedOp.PROD                         # 乘积
```

```
┌─────────────────────────────────────────────────────────────────┐
│                    num_warps配置指南                             │
├─────────────────────────────────────────────────────────────────┤
│                                                                 │
│  num_warps    适用场景              说明                        │
│  ─────────────────────────────────────────────────────────────  │
│  2            小规模计算            减少warp调度开销            │
│  4            通用场景              平衡性能和资源              │
│  8            大规模计算            提高并行度                  │
│  16           超大规模计算          需要足够的工作量            │
│                                                                 │
│  选择原则：                                                      │
│  • BLOCK_SIZE / num_warps 应该是32的倍数                        │
│  • 每个warp处理32个元素                                          │
│  • num_warps * 32 <= BLOCK_SIZE                                  │
│                                                                 │
└─────────────────────────────────────────────────────────────────┘
```
